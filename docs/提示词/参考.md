# 内容描述
BPE 训练的核心循环非常简单：统计 pair、选择最高频 pair、执行合并，再重复这一过程。
- 初始化词表：加入 256 个字节 token，以及作为硬边界的特殊 token。
- 预分词：按特殊 token 切分语料，再用 GPT-2 正则得到片段。后续合并只在片段内部进行。
- BPE 合并循环：每轮都完整扫描当前片段，执行以下步骤：
  1. 统计所有相邻 token pair 的出现次数；
  2. 选择频次最高的 pair，平局时取字节序更大的 pair；
  3. 将 pair 合并成新 token，加入词表并记录 merge；
  4. 再次完整遍历所有片段，替换本轮选中的 pair。

```python
# 语料预分词（此例按空白切）计数：{low:5, lower:2, widest:3, newest:6}
# 初始统计每个相邻 pair 的频次（片段用 UTF-8 字节表示）
#   es:9  st:9  we:8  lo:7  ow:7  ne:6  ew:6  wi:3  id:3  de:3  er:2

# 第 1 步：es 与 st 频次都是 9，平局取字典序更大的 st（s > e）
st = vocab[ord('s')] + vocab[ord('t')]   # 新 token b'st'，进词表 ID 257

# 第 2 步：把所有片段里的 (s, t) 合并成 st，并更新计数
#   low    -> low
#   lower  -> lower
#   widest -> w i d e (st)
#   newest -> n e w e (st)
# 下一轮又会选出新的最高频 pair（这里是 we），如此反复。
```

```python
# 预分词：用 GPT-2 的正则（来自 tiktoken，这里不 import，直接复制模式）把文本切成片段。
# 之后 BPE 合并只在每个片段内部进行，不会跨片段（也就不会把 "dog!" 和 "dog." 强行拼到一起）。

import regex as re

PAT = r"""'(?:[sdmt]|ll|ve|re)| ?\p{L}+| ?\p{N}+| ?[^\s\p{L}\p{N}]+|\s+(?!\S)|\s+"""

def preprocess_chunks(chunk_text: str, special_tokens_pattern: str) -> dict[str, int]:
    chunk_dict = {}
    # 先按特殊 token 切分（否则 "<|endoftext|>" 里的空格、竖线会被 GPT-2 正则拆散）
    for chunk1 in re.split(special_tokens_pattern, chunk_text):
        # 再对每块做 GPT-2 正则预分词，统计每个片段的出现次数
        for chunk2 in re.findall(PAT, chunk1):
            chunk_dict[chunk2] = chunk_dict.get(chunk2, 0) + 1
    return chunk_dict

# BPE 训练：每轮完整统计 pair，再完整遍历所有片段执行合并。
def train_bpe_reference(input_path, vocab_size, special_tokens):
    vocab = {i: bytes([i]) for i in range(256)}
    for i, st in enumerate(special_tokens):
        vocab[i + 256] = st.encode("utf-8")

    merges = []
    tokens = {}
    special_pattern = "|".join(re.escape(st) for st in special_tokens) if special_tokens else r"(?!)"
    with open(input_path, "r", encoding="utf-8") as f:
        for line in f:
            for chunk, count in preprocess_chunks(line.rstrip("\n"), special_pattern).items():
                ids = tuple(chunk.encode("utf-8"))
                tokens[ids] = tokens.get(ids, 0) + count

    while len(vocab) < vocab_size:
        # 1) 完整统计当前所有相邻 token pair
        stats = {}
        for ids, count in tokens.items():
            for p0, p1 in zip(ids, ids[1:]):
                stats[(p0, p1)] = stats.get((p0, p1), 0) + count
        if not stats:
            break

        # 2) 选择 pair，加入新 token
        p0, p1 = max(stats, key=lambda k: (stats[k], vocab[k[0]], vocab[k[1]]))
        new_id = len(vocab)
        vocab[new_id] = vocab[p0] + vocab[p1]
        merges.append((vocab[p0], vocab[p1]))

        # 3) 完整遍历 tokens，执行本轮合并
        new_tokens = {}
        for ids, count in tokens.items():
            merged = []
            i = 0
            while i < len(ids):
                if i + 1 < len(ids) and ids[i] == p0 and ids[i + 1] == p1:
                    merged.append(new_id)
                    i += 2
                else:
                    merged.append(ids[i])
                    i += 1
            merged = tuple(merged)
            new_tokens[merged] = new_tokens.get(merged, 0) + count
        tokens = new_tokens

    return vocab, merges
```
